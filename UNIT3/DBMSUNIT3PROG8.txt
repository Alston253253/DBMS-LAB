DECLARE
    v_num1 NUMBER;
    v_num2 NUMBER;
    v_result NUMBER;
BEGIN
    v_num1 := &num1;
    v_num2 := &num2;

    v_result := v_num1 / v_num2;

    DBMS_OUTPUT.PUT_LINE('Result = ' || v_result);

EXCEPTION
    WHEN ZERO_DIVIDE THEN
        DBMS_OUTPUT.PUT_LINE('Error: Division by zero is not allowed.');
END;
/
