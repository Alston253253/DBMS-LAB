CREATE TABLE EMP (
    EID NUMBER PRIMARY KEY,
    EName VARCHAR2(50),
    Deptno NUMBER,
    Deptname VARCHAR2(50),
    Gender VARCHAR2(10),
    Age NUMBER,
    BasicSal NUMBER(10,2)
);
DECLARE
    eid NUMBER;
    basic NUMBER;
    hra NUMBER;
    da NUMBER;
    medical NUMBER := 500;
    pf NUMBER;
    gross_salary NUMBER;
BEGIN
    eid := &eid;

    SELECT BasicSal
    INTO basic
    FROM EMP
    WHERE EID = eid;

    hra := basic * 15 / 100;
    da := basic * 50 / 100;
    pf := basic * 10 / 100;

    gross_salary := basic + (da + hra + medical) - pf;

    DBMS_OUTPUT.PUT_LINE('Gross Salary = Rs. ' || gross_salary);
END;
/
