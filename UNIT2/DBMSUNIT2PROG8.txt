DECLARE
    CURSOR c_customer IS
        SELECT *
        FROM CUSTOMER;

    v_customer CUSTOMER%ROWTYPE;
BEGIN
    OPEN c_customer;

    LOOP
        FETCH c_customer INTO v_customer;

        EXIT WHEN c_customer%NOTFOUND;

        DBMS_OUTPUT.PUT_LINE(
            v_customer.CustomerID || ' ' ||
            v_customer.CustomerName || ' ' ||
            v_customer.Address || ' ' ||
            v_customer.Phone
        );
    END LOOP;

    CLOSE c_customer;
END;
/
