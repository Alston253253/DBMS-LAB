DECLARE
    v_count NUMBER := 0;
BEGIN
    FOR emp_rec IN (
        SELECT EName, BasicSal
        FROM EMP
        ORDER BY BasicSal DESC
    )
    LOOP
        EXIT WHEN v_count = 3;

        v_count := v_count + 1;

        DBMS_OUTPUT.PUT_LINE(
            emp_rec.EName || ' ' || emp_rec.BasicSal
        );
    END LOOP;
END;
/
