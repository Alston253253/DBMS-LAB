DECLARE
    v_eid NUMBER;
    v_salary NUMBER;
BEGIN
    v_eid := &eid;
    v_salary := &salary;

    UPDATE EMPLOYEE
    SET Salary = v_salary
    WHERE EID = v_eid;

    IF SQL%ROWCOUNT > 0 THEN
        DBMS_OUTPUT.PUT_LINE('Employee salary updated successfully.');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Employee not found.');
    END IF;

    COMMIT;
END;
/
