DECLARE
    CURSOR c_emp IS
        SELECT EName, Deptname, BasicSal
        FROM EMP
        ORDER BY BasicSal DESC;

    v_count NUMBER := 0;
BEGIN
    OPEN c_emp;

    LOOP
        FETCH c_emp INTO v_name, v_dept, v_salary;

        EXIT WHEN c_emp%NOTFOUND OR v_count = 5;

        v_count := v_count + 1;

        DBMS_OUTPUT.PUT_LINE(
            v_name || ' ' || v_dept || ' ' || v_salary
        );
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Total records displayed: ' || v_count);

    CLOSE c_emp;
END;
/
